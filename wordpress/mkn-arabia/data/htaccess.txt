# BEGIN MKN ARABIA
# WordPress first, even if files from the old static site are still in public_html
DirectoryIndex index.php index.html
<IfModule mod_rewrite.c>
RewriteEngine On
# old static-site addresses -> WordPress addresses
RewriteRule ^index\.html$ / [R=301,L]
RewriteRule ^(services|pricing|work|testimonials|insights|about|contact|privacy|utm)\.html$ /$1/ [R=301,L]
RewriteRule ^insights/([a-z0-9-]+)\.html$ /insights/$1/ [R=301,L]
RewriteRule ^404\.html$ / [R=301,L]
RewriteRule ^sitemap\.xml$ /wp-sitemap.xml [R=301,L]
</IfModule>
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType text/css "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType image/webp "access plus 6 months"
  ExpiresByType image/png "access plus 6 months"
  ExpiresByType image/jpeg "access plus 6 months"
</IfModule>
AddType font/woff2 .woff2
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
</IfModule>
# END MKN ARABIA
